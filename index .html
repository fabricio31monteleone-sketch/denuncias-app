async function enviarDenuncia() { const tipo = document.getElementById('tipo').value; const detalle = document.getElementById('detalle').value; const ubicacion = document.getElementById('ubicacion').value; const btnEnviar = document.getElementById('btnEnviar'); if (!tipo || !detalle) { alert('Por favor complete todos los campos requeridos.'); return; } btnEnviar.innerText = 'Enviando...'; btnEnviar.disabled = true; // Lista de correos para copia oculta (BCC) const bccEmails = [ 'sehacienda@mda.gob.ar', 'secsalud@mda.gob.ar', 'produccioncya@mda.gob.ar', 'seleccionydesarrollo@mda.gob.ar', 'legalytecnica@mda.gob.ar', 'lalevenet@gmail.com' ]; try { const adjuntosFormatted = listaArchivos.map(item => ({ filename: item.filename, path: item.path })); const res = await fetch('/api/denuncia', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ tipo, detalle, ubicacion, adjuntos: adjuntosFormatted, bcc: bccEmails // Envíamos la lista de copias ocultas al servidor }) }); const data = await res.json(); if (data.success) { alert('¡Reclamo enviado con éxito!'); document.getElementById('formMunicipal').reset(); listaArchivos = []; actualizarVistaPrevia(); } else { alert('Error al enviar: ' + data.message); } } catch (err) { alert('Error de conexión con el servidor.'); } finally { btnEnviar.innerText = 'Enviar Reclamo'; btnEnviar.disabled = false; } }
